    function recordBall(type, val) {
      if (matchFinished) return;

      const inn = getActiveInnings();
      const maxBalls = totalOversLimit * 6;
      const maxWickets = isSuperOver ? 2 : 10;
      if (inn.legalBalls &gt;= maxBalls || inn.wickets &gt;= maxWickets) return;

      const bowlerStats = getBowlerStats(currentBowlerName, inn);

      // Check if bowler quota is exceeded
      if (checkBowlerQuotaReached(bowlerStats) &amp;&amp; (inn.legalBalls % 6 === 0)) {
      showToast(`${currentBowlerName} completed maximum quota of ${maxBowlerOvers} overs. Please choose another bowler!`, &#39;⚠️&#39;, &#39;border-amber-500/80&#39;, &#39;text-amber-200&#39;);
        changeBowlerBtn.click();
        return;
      }

      const snapshot = {
        currentInnings,
        matchFormat,
        totalOversLimit,
        maxBowlerOvers,
        targetScore,
        matchFinished,
        isSuperOver,
        innSnapshot: JSON.parse(JSON.stringify(inn)),
        striker: JSON.parse(JSON.stringify(striker)),
        nonStriker: JSON.parse(JSON.stringify(nonStriker)),
        currentBowlerName,
        batsmanOutCount,
        type,
        val
      };

      let ballRuns = 0;
      let isLegal = true;

      if (type === &#39;run&#39;) {
        const r = parseInt(val);
        ballRuns = r;
        inn.runs += r;
        inn.legalBalls++;

        striker.runs += r;
        striker.balls++;

        if (r === 4) { inn.fours++; striker.fours++; playFourSound(); }
        else if (r === 6) { inn.sixes++; striker.sixes++; playSixSound(); }
        else if (r &gt; 0) { playTapSound(); }

        bowlerStats.legalBalls++;
        bowlerStats.runs += r;
        currentOverBalls.push(r);

        if (r === 1 || r === 3) swapStrike();
      } else if (type === &#39;wicket&#39;) {
        inn.wickets++;
        inn.legalBalls++;
        striker.balls++;

        bowlerStats.legalBalls++;
        bowlerStats.wickets++;
        currentOverBalls.push(0);

        batsmanOutCount++;
        playWicketSound();

        striker = {
          name: `Batter ${batsmanOutCount + 2}`,
          runs: 0,
          balls: 0,
          fours: 0,
          sixes: 0
        };
      } else if (type === &#39;extra&#39;) {
        if (val === &#39;Wd&#39; || val === &#39;Nb&#39;) {
          inn.runs += 1;
          inn.extras += 1;
          bowlerStats.runs += 1;
          currentOverBalls.push(1);
          isLegal = false;
          playTapSound();
        } else if (val === &#39;B&#39;) {
          inn.runs += 1;
          inn.extras += 1;
          inn.legalBalls++;
          striker.balls++;
          bowlerStats.legalBalls++;
          currentOverBalls.push(0);
          isLegal = true;
          playTapSound();
          swapStrike();
        }
      }

      snapshot.isLegal = isLegal;
      inn.ballHistory.push(snapshot);

      // Over Completion check
      if (isLegal &amp;&amp; inn.legalBalls % 6 === 0) {
        swapStrike();

        if (currentOverBalls.length &gt;= 6 &amp;&amp; currentOverBalls.every(r =&gt; r === 0)) {
          bowlerStats.maidens++;
        }
        currentOverBalls = [];

        // Check if bowler finished quota or prompt next bowler
        setTimeout(() =&gt; {
          if (!matchFinished &amp;&amp; inn.legalBalls &lt; maxBalls &amp;&amp; inn.wickets &lt; maxWickets) {
            changeBowlerBtn.click();
          }
        }, 300);
      }

      // Check Innings End
      if (currentInnings === 1 || currentInnings === &#39;SUPER_1&#39;) {
        if (inn.legalBalls &gt;= maxBalls || inn.wickets &gt;= maxWickets) {
          targetScore = inn.runs + 1;
          inningsBreakSummary.textContent = `${inn.battingTeam} scored ${inn.runs}/${inn.wickets} in ${Math.floor(inn.legalBalls / 6)}.${inn.legalBalls % 6} overs`;
          breakTargetDisplay.textContent = `${targetScore} Runs (${Math.floor(targetScore / totalOversLimit * 10) / 10} RPO)`;
          start2ndInningsBtn.textContent = `Start ${isSuperOver ? &#39;Super Over&#39; : &#39;2nd Innings&#39;} (${isSuperOver ? teamAName : teamBName} Batting)`;
          inningsBreakModal.classList.remove(&#39;hidden&#39;);
          playVictoryFanfare();
        }
      } else {
        // 2nd innings or Super Over 2
        if (inn.runs &gt;= targetScore) {
          playVictoryFanfare();
        }
      }

      saveActiveMatch();
      render();
    }

    function undoLastBall() {
      const inn = getActiveInnings();
      if (inn.ballHistory.length === 0) return;

      const last = inn.ballHistory.pop();
      currentInnings = last.currentInnings;
      matchFormat = last.matchFormat;
      totalOversLimit = last.totalOversLimit;
      maxBowlerOvers = last.maxBowlerOvers;
      targetScore = last.targetScore;
      matchFinished = false;
      isSuperOver = last.isSuperOver;

      if (currentInnings === 1 || currentInnings === &#39;SUPER_1&#39;) innings1 = last.innSnapshot;
      else innings2 = last.innSnapshot;

      striker = last.striker;
      nonStriker = last.nonStriker;
      currentBowlerName = last.currentBowlerName;
      batsmanOutCount = last.batsmanOutCount;

      saveActiveMatch();
      render();
    }

    function renderHistoryList() {
      historyMatchesList.innerHTML = &#39;&#39;;
      let history = [];
      try {
        const raw = localStorage.getItem(MATCHES_HISTORY_KEY);
        if (raw) history = JSON.parse(raw);
      } catch (e) {}

      if (history.length === 0) {
        historyMatchesList.innerHTML = &#39;&lt;div class=&quot;text-center py-6 text-slate-500 text-xs&quot;&gt;No saved matches found.&lt;/div&gt;&#39;;
        return;
      }

      history.forEach((m, idx) =&gt; {
        const card = document.createElement(&#39;div&#39;);
        const isCurrentActive = (m.id === window.matchId);
        const dateStr = new Date(m.updatedAt).toLocaleDateString([], { month: &#39;short&#39;, day: &#39;numeric&#39;, hour: &#39;2-digit&#39;, minute: &#39;2-digit&#39; });

        const score1 = `${m.innings1.runs}/${m.innings1.wickets} (${Math.floor(m.innings1.legalBalls/6)}.${m.innings1.legalBalls%6} ov)`;
        const is2nd = (m.currentInnings === 2 || m.currentInnings === &#39;SUPER_2&#39;);
        const score2 = is2nd 
          ? `${m.innings2.runs}/${m.innings2.wickets} (${Math.floor(m.innings2.legalBalls/6)}.${m.innings2.legalBalls%6} ov)` 
          : &#39;Yet to bat&#39;;

        card.className = `p-2.5 rounded-xl border text-xs flex flex-col space-y-1.5 transition-all ${isCurrentActive ? &#39;bg-blue-950/80 border-blue-500 shadow-md&#39; : &#39;bg-slate-800/80 border-slate-700&#39;}`;
        card.innerHTML = `
          &lt;div class=&quot;flex justify-between items-center text-[10px]&quot;&gt;
            &lt;span class=&quot;font-bold text-amber-300&quot;&gt;${m.isSuperOver ? &#39;⚡ Super Over&#39; : `${m.matchFormat} (${m.totalOversLimit} ov)`}&lt;/span&gt;
            &lt;span class=&quot;text-slate-400&quot;&gt;${dateStr}&lt;/span&gt;
          &lt;/div&gt;
          &lt;div class=&quot;flex justify-between items-center font-bold text-slate-100&quot;&gt;
            &lt;span&gt;${m.teamAName} vs ${m.teamBName}&lt;/span&gt;
            &lt;span class=&quot;text-emerald-400&quot;&gt;${score1}&lt;/span&gt;
          &lt;/div&gt;
          &lt;div class=&quot;flex justify-between items-center text-[11px] text-slate-400&quot;&gt;
            &lt;span&gt;${is2nd ? `2nd Inn (${m.teamBName}): ${score2}` : &#39;1st Innings in progress&#39;}&lt;/span&gt;
            &lt;span class=&quot;text-[9px] px-1.5 py-0.5 rounded font-bold ${m.matchFinished ? &#39;bg-emerald-900/60 text-emerald-300&#39; : &#39;bg-blue-900/60 text-blue-300&#39;}&quot;&gt;
              ${m.matchFinished ? &#39;Completed&#39; : &#39;Paused&#39;}
            &lt;/span&gt;
          &lt;/div&gt;
          &lt;button 
            type=&quot;button&quot; 
            class=&quot;resume-btn mt-1 w-full py-1 bg-blue-600 hover:bg-blue-500 active:scale-95 text-white font-bold text-[11px] rounded-lg transition-transform&quot;
            data-index=&quot;${idx}&quot;
          &gt;
            ${isCurrentActive ? &#39;▶ Keep Playing Current Match&#39; : &#39;▶ Resume &amp; Open Match&#39;}
          &lt;/button&gt;
        `;
        historyMatchesList.appendChild(card);
      });
    }
